""" Skin case bot (виртуальная валюта, без реальных выплат). Фичи: велком-кейс, кейсы с шансами, инвентарь, продажа скинов, ежедневный бонус, рефералы, топ игроков. Установка: pip install aiogram aiosqlite Запуск: BOT_TOKEN=123:abc python skin_bot.py """ import asyncio import os import random import time import aiosqlite from aiogram import Bot, Dispatcher, F from aiogram.filters import Command, CommandObject from aiogram.types import CallbackQuery, InlineKeyboardButton as Btn, InlineKeyboardMarkup as Kb, Message DB = "skins.db" REF_BONUS = 50 DAILY_BONUS = 25 # Редкости: (название, эмодзи, вес выпадения, цена продажи min-max) RARITY = { "consumer": ("Ширпотреб", "⚪", 50, (5, 15)), "milspec": ("Армейское", "🔵", 30, (20, 60)), "restricted": ("Запрещённое", "🟣", 13, (80, 200)), "classified": ("Засекреченное", "🌸", 5, (250, 600)), "covert": ("Тайное", "🔴", 1.8, (800, 2000)), "gold": ("Редкое золото", "🟡", 0.2, (3000, 8000)), } SKINS = { "consumer": ["P250 | Sand Dune", "MP9 | Storm", "Nova | Predator"], "milspec": ["MAC-10 | Neon Rider", "Glock-18 | Water Elemental", "SG 553 | Cyrex"], "restricted": ["AK-47 | Redline", "M4A4 | Desolate Space", "USP-S | Cortex"], "classified": ["AWP | Asiimov", "M4A1-S | Hyper Beast", "Desert Eagle | Blaze"], "covert": ["AK-47 | Fire Serpent", "AWP | Dragon Lore", "M4A4 | Howl"], "gold": ["★ Karambit | Fade", "★ Butterfly Knife | Doppler", "★ Sport Gloves | Vice"], } # Кейсы: id -> (название, цена в кристаллах) CASES = { "welcome": ("🎁 Велком-кейс", 0), "basic": ("📦 Базовый кейс", 100), "elite": ("💎 Элитный кейс", 400), } # Элитный кейс смещает шансы в сторону редких ELITE_MULT = {"consumer": 0, "milspec": 0.5, "restricted": 1.5, "classified": 2.5, "covert": 3, "gold": 4} def roll(case_id: str): weights = [] for key, (_, _, w, _) in RARITY.items(): mult = ELITE_MULT[key] if case_id == "elite" else 1 weights.append(w * mult) key = random.choices(list(RARITY), weights=weights)[0] name = random.choice(SKINS[key]) lo, hi = RARITY[key][3] return key, name, random.randint(lo, hi) async def init_db(): async with aiosqlite.connect(DB) as db: await db.executescript( """ CREATE TABLE IF NOT EXISTS users( id INTEGER PRIMARY KEY, name TEXT, crystals INTEGER DEFAULT 0, welcome_used INTEGER DEFAULT 0, last_daily INTEGER DEFAULT 0, referrer INTEGER, opened INTEGER DEFAULT 0); CREATE TABLE IF NOT EXISTS inventory( id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, rarity TEXT, skin TEXT, price INTEGER); """ ) await db.commit() async def get_user(db, uid): cur = await db.execute("SELECT crystals, welcome_used, last_daily, opened FROM users WHERE id=?", (uid,)) return await cur.fetchone() def main_kb(): return Kb(inline_keyboard=[ [Btn(text="📦 Кейсы", callback_data="cases"), Btn(text="🎒 Инвентарь", callback_data="inv")], [Btn(text="🎁 Бонус дня", callback_data="daily"), Btn(text="🏆 Топ", callback_data="top")], [Btn(text="👥 Пригласить друзей", callback_data="ref")], ]) bot = Bot(os.environ["BOT_TOKEN"]) dp = Dispatcher() @dp.message(Command("start")) async def start(m: Message, command: CommandObject): uid = m.from_user.id async with aiosqlite.connect(DB) as db: if not await get_user(db, uid): ref = int(command.args) if command.args and command.args.isdigit() and int(command.args) != uid else None await db.execute("INSERT INTO users(id, name, referrer) VALUES(?,?,?)", (uid, m.from_user.full_name, ref)) if ref and await get_user(db, ref): await db.execute("UPDATE users SET crystals=crystals+? WHERE id=?", (REF_BONUS, ref)) try: await bot.send_message(ref, f"👥 По твоей ссылке пришёл друг! +{REF_BONUS} 💎") except Exception: pass await db.commit() await m.answer( "🔥 Добро пожаловать!\nОткрывай кейсы, собирай скины, продавай их за кристаллы.\n" "Тебя ждёт бесплатный велком-кейс 🎁", reply_markup=main_kb(), parse_mode="HTML", ) @dp.callback_query(F.data == "menu") async def menu(c: CallbackQuery): async with aiosqlite.connect(DB) as db: u = await get_user(db, c.from_user.id) await c.message.edit_text(f"💎 Баланс: {u[0]}\nВыбирай действие:", reply_markup=main_kb(), parse_mode="HTML") @dp.callback_query(F.data == "cases") async def cases(c: CallbackQuery): rows = [[Btn(text=f"{n} — {p} 💎" if p else f"{n} — бесплатно", callback_data=f"open:{cid}")] for cid, (n, p) in CASES.items()] rows.append([Btn(text="⬅️ Назад", callback_data="menu")]) await c.message.edit_text("Выбери кейс:", reply_markup=Kb(inline_keyboard=rows)) @dp.callback_query(F.data.startswith("open:")) async def open_case(c: CallbackQuery): cid = c.data.split(":")[1] name, price = CASES[cid] uid = c.from_user.id async with aiosqlite.connect(DB) as db: crystals, welcome_used, _, _ = await get_user(db, uid) if cid == "welcome": if welcome_used: return await c.answer("Велком-кейс уже открыт!", show_alert=True) await db.execute("UPDATE users SET welcome_used=1 WHERE id=?", (uid,)) elif crystals < price: return await c.answer("Не хватает кристаллов 💎", show_alert=True) else: await db.execute("UPDATE users SET crystals=crystals-? WHERE id=?", (price, uid)) await db.commit() await c.answer() # "Анимация" прокрутки for _ in range(4): k = random.choice(list(RARITY)) await c.message.edit_text(f"🎰 Крутим {name}...\n{RARITY[k][1]} {random.choice(SKINS[k])}") await asyncio.sleep(0.6) key, skin, val = roll(cid) async with aiosqlite.connect(DB) as db: await db.execute("INSERT INTO inventory(user_id, rarity, skin, price) VALUES(?,?,?,?)", (uid, key, skin, val)) await db.execute("UPDATE users SET opened=opened+1 WHERE id=?", (uid,)) await db.commit() label, emoji = RARITY[key][0], RARITY[key][1] kb = Kb(inline_keyboard=[ [Btn(text="🔁 Ещё раз", callback_data=f"open:{cid}"), Btn(text="🎒 Инвентарь", callback_data="inv")], [Btn(text="⬅️ Меню", callback_data="menu")], ]) await c.message.edit_text( f"{emoji} {skin}\nРедкость: {label}\nЦена: {val} 💎", reply_markup=kb, parse_mode="HTML") @dp.callback_query(F.data == "inv") async def inventory(c: CallbackQuery): async with aiosqlite.connect(DB) as db: cur = await db.execute( "SELECT id, rarity, skin, price FROM inventory WHERE user_id=? ORDER BY price DESC LIMIT 10", (c.from_user.id,)) items = await cur.fetchall() if not items: return await c.message.edit_text("Инвентарь пуст. Открой кейс!", reply_markup=Kb( inline_keyboard=[[Btn(text="📦 Кейсы", callback_data="cases")]])) rows = [[Btn(text=f"{RARITY[r][1]} {s} · продать за {p}💎", callback_data=f"sell:{i}")] for i, r, s, p in items] rows.append([Btn(text="💰 Продать всё", callback_data="sellall"), Btn(text="⬅️ Назад", callback_data="menu")]) await c.message.edit_text("🎒 Твои лучшие скины (топ-10):", reply_markup=Kb(inline_keyboard=rows)) @dp.callback_query(F.data.startswith("sell:")) async def sell(c: CallbackQuery): item_id = int(c.data.split(":")[1]) async with aiosqlite.connect(DB) as db: cur = await db.execute("SELECT price FROM inventory WHERE id=? AND user_id=?", (item_id, c.from_user.id)) row = await cur.fetchone() if row: await db.execute("DELETE FROM inventory WHERE id=?", (item_id,)) await db.execute("UPDATE users SET crystals=crystals+? WHERE id=?", (row[0], c.from_user.id)) await db.commit() await c.answer(f"+{row[0]} 💎") await inventory(c) @dp.callback_query(F.data == "sellall") async def sell_all(c: CallbackQuery): async with aiosqlite.connect(DB) as db: cur = await db.execute("SELECT COALESCE(SUM(price),0) FROM inventory WHERE user_id=?", (c.from_user.id,)) total = (await cur.fetchone())[0] await db.execute("DELETE FROM inventory WHERE user_id=?", (c.from_user.id,)) await db.execute("UPDATE users SET crystals=crystals+? WHERE id=?", (total, c.from_user.id)) await db.commit() await c.answer(f"Продано на {total} 💎", show_alert=True) await inventory(c) @dp.callback_query(F.data == "daily") async def daily(c: CallbackQuery): async with aiosqlite.connect(DB) as db: _, _, last, _ = await get_user(db, c.from_user.id) now = int(time.time()) if now - last < 86400: left = (86400 - (now - last)) // 3600 return await c.answer(f"Приходи через ~{left} ч.", show_alert=True) await db.execute("UPDATE users SET crystals=crystals+?, last_daily=? WHERE id=?", (DAILY_BONUS, now, c.from_user.id)) await db.commit() await c.answer(f"+{DAILY_BONUS} 💎 Бонус получен!", show_alert=True) @dp.callback_query(F.data == "top") async def top(c: CallbackQuery): async with aiosqlite.connect(DB) as db: cur = await db.execute("SELECT name, opened FROM users ORDER BY opened DESC LIMIT 10") rows = await cur.fetchall() text = "🏆 Топ по открытым кейсам\n" + "\n".join(f"{i+1}. {n} — {o}" for i, (n, o) in enumerate(rows)) await c.message.edit_text(text, parse_mode="HTML", reply_markup=Kb(inline_keyboard=[[Btn(text="⬅️ Назад", callback_data="menu")]])) @dp.callback_query(F.data == "ref") async def ref(c: CallbackQuery): me = await bot.get_me() link = f"https://t.me/{me.username}?start={c.from_user.id}" await c.message.edit_text(f"👥 Приглашай друзей и получай по {REF_BONUS} 💎 за каждого:\n{link}", reply_markup=Kb(inline_keyboard=[[Btn(text="⬅️ Назад", callback_data="menu")]])) async def main(): await init_db() await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
